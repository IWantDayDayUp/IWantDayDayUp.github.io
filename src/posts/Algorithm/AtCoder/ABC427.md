```cpp
#include <bits/stdc++.h>

using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    int ans = 0;
    int cur = 1;
    while (N--)
    {
        string s = to_string(cur);
        int temp = 0;
        for (auto c : s)
        {
            temp += c - '0';
        }
        ans += temp;
        cur = ans;
    }

    cout << ans << endl;

    return 0;
}
```

## 📌 C - Bipartize{#C}

![427C](https://github.com/IWantDayDayUp/picx-images-hosting/raw/master/ABC/427C.1zincxlzh8.png)

::: tip
A coloring is representing by a bitmask `mask` of length $N$

- Bit $i = 0$: vertex $i$ is in color `A`
- Bit $i = 1$: vertex $i$ is in color `B`

For each unordered pair $(i, j)$ with $i < j$, if they have the same bit in `mask`, then any edge between them would be same-color and must be deleted

- Adds `adj[i][j]` to `cnt` ($0$ if no edge, $1$ if there is)
:::

```cpp
#include <bits/stdc++.h>

using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, M;
    cin >> N >> M;

    vector<vector<int>> adj(N, vector<int>(N));
    for (int i = 0; i < M; i++)
    {
        int u, v;
        cin >> u >> v;
        u--;
        v--;

        adj[u][v] = adj[v][u] = 1;
    }

    int ans = 1 << 30;
    for (int mask = 0; mask < (1 << N); mask++)
    {
        int cnt = 0;

        for (int i = 0; i < N; i++)
        {
            for (int j = i + 1; j < N; j++)
            {
                if ((mask >> i & 1) == (mask >> j & 1))
                {
                    cnt += adj[i][j];
                }
            }
        }

        ans = min(ans, cnt);
    }

    cout << ans << endl;

    return 0;
}
```

## 📌 D - The Simple Game{#D}

![427D](https://github.com/IWantDayDayUp/picx-images-hosting/raw/master/ABC/427D.2rviuo2l7h.png)

::: tip
In this kind of **game theory** tasks, it is often effective to backtrack the winners from the final states
:::

::: tip
Used a **layered dynamic programming(DP)** from the end to the start

> $dp_{i, j}$: boolean value, representing whether Alice will always win

- $i$: number of moves still to play, because this value tells us whose turn it is
- $j$: the piece is currently at vertex $j$
- $dp_{i, j} = true$: means Alice will win from this state if both play optimally

Transitions:

For $i = 0, 1, 2, ..., 2K$, Let $u \rightarrow v$ denote an outgoing edge

- If it is Alice's turn, she chooses a move that helps her
$$dp_{i, u} = OR \, over \, all \, edges (u \rightarrow v) \, of \, dp_{i - 1, v}$$
- If it is Bob's turn, he tries to force Alice to lose
$$dp_{i, u} = AND \, over \, all \, edges (u \rightarrow v) \, of \, dp_{i - 1, v}$$
:::

```cpp
#include <bits/stdc++.h>
using namespace std;

void solve()
{
    int N, M, K;
    cin >> N >> M >> K;

    string S;
    cin >> S;

    vector<vector<int>> adj(N);
    for (int i = 0; i < M; i++)
    {
        int u, v;
        cin >> u >> v;
        u--;
        v--;

        adj[u].push_back(v);
    }

    // t=0
    vector<char> dp(N, 0);
    for (int u = 0; u < N; u++)
        dp[u] = (S[u] == 'A');

    for (int t = 1; t <= 2 * K; t++)
    {
        vector<char> ndp(N);
        if (t % 2 == 0)
        { // Alice
            for (int u = 0; u < N; u++)
            {
                char ok = 0;
                for (auto v : adj[u])
                {
                    if (dp[v])
                    {
                        ok = 1;
                        break;
                    }
                }
                ndp[u] = ok;
            }
        }
        else
        { // Bob
            for (int u = 0; u < N; u++)
            {
                char ok = 1;
                for (auto v : adj[u])
                {
                    if (!dp[v])
                    {
                        ok = 0;
                        break;
                    }
                }
                ndp[u] = ok;
            }
        }
        dp.swap(ndp);
    }

    cout << (dp[0] ? "Alice" : "Bob") << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--)
    {
        solve();
    }

    return 0;
}

```
